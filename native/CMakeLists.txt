cmake_minimum_required(VERSION 3.20)
project(voxmask_audio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(voxmask_audio STATIC audio/audio.cpp audio/buffer_check.cpp)
target_include_directories(voxmask_audio PUBLIC include)
target_link_libraries(voxmask_audio PUBLIC Threads::Threads)
if(APPLE)
 target_link_libraries(voxmask_audio PUBLIC "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox")
elseif(UNIX)
 target_link_libraries(voxmask_audio PUBLIC dl pthread m)
endif()
include(CTest)
if(BUILD_TESTING)
 find_package(Threads REQUIRED)
 add_executable(processor_test tests/processor_test.cpp)
 target_link_libraries(processor_test PRIVATE Threads::Threads)
 add_test(NAME native_processor COMMAND processor_test)
 add_executable(frame_ring_test tests/frame_ring_test.cpp audio/buffer_check.cpp)
 target_link_libraries(frame_ring_test PRIVATE Threads::Threads)
 add_test(NAME native_frame_ring COMMAND frame_ring_test)
 add_executable(worker_pipeline_test tests/worker_pipeline_test.cpp)
 target_link_libraries(worker_pipeline_test PRIVATE Threads::Threads)
 add_test(NAME native_worker_pipeline COMMAND worker_pipeline_test)
 add_executable(producer_protocol_test tests/producer_protocol_test.cpp)
 target_link_libraries(producer_protocol_test PRIVATE Threads::Threads)
 add_test(NAME native_producer_protocol COMMAND producer_protocol_test)
 add_executable(producer_mailbox_test tests/producer_mailbox_test.cpp)
 target_link_libraries(producer_mailbox_test PRIVATE Threads::Threads)
 add_test(NAME native_producer_mailbox COMMAND producer_mailbox_test)
 add_executable(abi_test tests/abi_test.cpp)
 target_link_libraries(abi_test PRIVATE voxmask_audio)
 add_test(NAME native_abi COMMAND abi_test)
endif()
